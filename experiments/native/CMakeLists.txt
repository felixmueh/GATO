cmake_minimum_required(VERSION 3.24)
project(gato_experiment_modules LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(KNOTS "32" CACHE STRING "State knot counts")
option(GATO_EXPERIMENT_TELEMETRY "Retain native SQP/PCG work counters" ON)
if(NOT GATO_SOURCE_DIR OR NOT EXISTS "${GATO_SOURCE_DIR}/python/bindings.cu")
  message(FATAL_ERROR "Set GATO_SOURCE_DIR to the pinned GATO checkout")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
execute_process(COMMAND "${Python3_EXECUTABLE}" -m pybind11 --cmakedir
                OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE
                COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED)
foreach(n IN LISTS KNOTS)
  set(module bsqpN${n}_tiago_right)
  pybind11_add_module(${module} SHARED "${GATO_SOURCE_DIR}/python/bindings.cu")
  target_include_directories(${module} PRIVATE "${GATO_SOURCE_DIR}/gato")
  target_compile_definitions(${module} PRIVATE KNOT_POINTS=${n} PLANT_TIAGO_RIGHT=1)
  if(GATO_EXPERIMENT_TELEMETRY)
    target_compile_definitions(${module} PRIVATE GATO_EXPERIMENT_TELEMETRY=1)
  endif()
  target_compile_options(${module} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-use_fast_math;-O3>)
  set_target_properties(${module} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
endforeach()
