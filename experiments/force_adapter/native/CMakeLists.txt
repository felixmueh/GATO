cmake_minimum_required(VERSION 3.24)
project(force_runtime_qualification LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
get_filename_component(GATO_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
execute_process(COMMAND "${Python3_EXECUTABLE}" -m pybind11 --cmakedir OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(bsqpN32_tiago_right SHARED "${GATO_SOURCE_DIR}/python/bindings.cu")
add_executable(force_probe force_probe.cu)
foreach(target bsqpN32_tiago_right force_probe)
  target_include_directories(${target} PRIVATE "${GATO_SOURCE_DIR}/gato")
  target_compile_definitions(${target} PRIVATE KNOT_POINTS=32 PLANT_TIAGO_RIGHT=1)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-use_fast_math;-O3>)
endforeach()
set_target_properties(bsqpN32_tiago_right PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
add_executable(cuda_access_smoke "${GATO_SOURCE_DIR}/tests/cpp/cuda_access_smoke.cu")
target_include_directories(cuda_access_smoke PRIVATE "${GATO_SOURCE_DIR}/gato")
