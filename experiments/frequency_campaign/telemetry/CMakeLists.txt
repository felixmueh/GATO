# Root builds this only after source/protocol review, in a build-owned snapshot.
cmake_minimum_required(VERSION 3.24)
project(ExperimentTelemetryFixtures LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "Frozen telemetry comparison requires Release")
endif()
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES OR CMAKE_CUDA_ARCHITECTURES STREQUAL "native")
  message(FATAL_ERROR "Pass the reviewed explicit CUDA architecture")
endif()
get_filename_component(GATO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
option(GATO_EXPERIMENT_TELEMETRY "Record existing SQP/PCG stop paths" OFF)
add_executable(direct_pcg direct_pcg.cu)
target_include_directories(direct_pcg PRIVATE "${GATO_ROOT}/gato")
target_compile_definitions(direct_pcg PRIVATE KNOT_POINTS=32 PLANT_TIAGO_RIGHT=1)
if(GATO_EXPERIMENT_TELEMETRY)
  target_compile_definitions(direct_pcg PRIVATE GATO_EXPERIMENT_TELEMETRY=1)
endif()
target_compile_options(direct_pcg PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-use_fast_math;-Xptxas=-v>)
