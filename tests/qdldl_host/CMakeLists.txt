cmake_minimum_required(VERSION 3.18)
project(qdldl_host_check LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(GATO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${GATO_ROOT}/cmake/qdldl.cmake")
add_executable(qdldl_host_check known_system.cpp)
target_include_directories(qdldl_host_check PRIVATE "${GATO_ROOT}/gato")
target_link_libraries(qdldl_host_check PRIVATE qdldlstatic)
enable_testing()
add_test(NAME qdldl_host_check COMMAND qdldl_host_check)
